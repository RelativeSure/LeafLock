## Environment Variables

### Required Configuration

| Variable | Description | Example |
|----------|-------------|---------|
| `CLERK_PUBLISHABLE_KEY` | Clerk publishable key for backend | `pk_test_abc123...` |
| `CLERK_SECRET_KEY` | Clerk secret key for backend | `sk_test_xyz789...` |
| `VITE_CLERK_PUBLISHABLE_KEY` | Clerk publishable key for frontend | `pk_test_abc123...` |

### Optional Configuration

| Variable | Description | Default | Example |
|----------|-------------|---------|---------|
| `VITE_ENABLE_REGISTRATION` | Enable/disable user registration | `true` | `true` or `false` |
| `CLERK_SIGN_IN_URL` | Custom sign-in URL | `/login` | `/auth/signin` |
| `CLERK_SIGN_UP_URL` | Custom sign-up URL | `/register` | `/auth/signup` |
| `CLERK_AFTER_SIGN_IN_URL` | Redirect after sign in | `/` | `/dashboard` |
| `CLERK_AFTER_SIGN_UP_URL` | Redirect after sign up | `/` | `/onboarding` |
| `CLERK_AFTER_SIGN_OUT_URL` | Redirect after sign out | `/login` | `/` |

### Custom Authentication Flow

```tsx title="Custom Authentication Hook"
import { useSignIn, useSignUp } from '@clerk/clerk-react'

function CustomAuthFlow() {
  const { signIn, isLoaded: signInLoaded } = useSignIn()
  const { signUp, isLoaded: signUpLoaded } = useSignUp()
  
  const handleCustomSignIn = async (email: string, password: string) => {
    if (!signInLoaded) return
    
    try {
      await signIn.create({
        identifier: email,
        password: password,
      })
      
      // Handle success
      router.push('/')
    } catch (error) {
      // Handle error
      console.error('Sign in failed:', error)
    }
  }
  
  return (
    <form onSubmit={handleSubmit}>
      {/* Custom form implementation */}
    </form>
  )
}
```

### Route Configuration

```go title="backend/routes.go"
func setupRoutes(app *fiber.App, db *pgxpool.Pool, rdb *redis.Client, config *appconfig.Config, startTime time.Time, readyState *appserver.ReadyState) {
    // Authentication configuration
    authConfig := auth.Config{
        MaxAttempts:   config.MaxLoginAttempts,
        LockoutDuration: config.LockoutDuration,
        SessionDuration: config.SessionDuration,
        ClerkEnabled:  config.ClerkSecretKey != "",
        JWTEnabled:    len(config.JWTSecret) > 0,
    }
    
    // Apply authentication middleware based on configuration
    if authConfig.ClerkEnabled || authConfig.JWTEnabled {
        // Use dual authentication middleware
        protected := api.Group("", authHandler.DualAuthMiddleware)
        
        // Conditional middleware based on auth type
        if authConfig.ClerkEnabled {
            // Clerk-specific routes
            api.Post("/auth/clerk/webhook", rateLimits.StandardLimiter, authHandler.ClerkWebhook)
        }
        
        if authConfig.JWTEnabled {
            // JWT-specific routes during transition
            api.Post("/auth/jwt/refresh", rateLimits.StandardLimiter, authHandler.JWTRefresh)
        }
    }
}
```

## Security Configuration

### Rate Limiting

```go title="Rate Limiting Configuration"
// Backend rate limiting
func (h *Handler) setupRateLimiting() {
    rateLimits := middleware.RateLimitConfig{
        AuthEndpoints: middleware.RateLimit{
            Requests: 10,
            Window:   5 * time.Minute,
            KeyFunc: func(c *fiber.Ctx) string {
                return getClientIP(c) + ":auth"
            },
        },
        ClerkSpecific: middleware.RateLimit{
            Requests: 20,
            Window:   1 * time.Minute,
            KeyFunc: func(c *fiber.Ctx) string {
                return getClientIP(c) + ":clerk"
            },
        },
    }
    
    // Apply rate limiting middleware
    app.Use(middleware.RateLimitMiddleware(rateLimits))
}
```

### Session Security

```go title="Session Security Configuration"
func (h *Handler) createSecureSession(ctx context.Context, userID uuid.UUID, authType string) (*Session, error) {
    session := &Session{
        ID:        uuid.New(),
        UserID:    userID,
        AuthType:  authType,
        IPAddress: getClientIP(ctx),
        UserAgent: getUserAgent(ctx),
        ExpiresAt: time.Now().Add(h.config.SessionDuration),
        CreatedAt: time.Now(),
        Secure:    true,
        HttpOnly:  true,
        SameSite:  "strict",
    }
    
    // Additional security measures
    if authType == "clerk" {
        session.TokenHash = generateSecureHash(session.ID.String())
        session.EncryptionKey = deriveSessionKey(session.ID, userID)
    }
    
    return session, nil
}
```

## Monitoring and Analytics

### Authentication Metrics

```go title="Metrics Configuration"
// Set up authentication metrics
authMetrics := prometheus.NewCounterVec(
    prometheus.CounterOpts{
        Name: "leaflock_auth_attempts_total",
        Help: "Total number of authentication attempts",
    },
    []string{"auth_type", "success"},
)

// Record authentication events
func recordAuthEvent(authType string, success bool) {
    authMetrics.WithLabelValues(authType, strconv.FormatBool(success)).Inc()
}
```

### Logging Configuration

```go title="Logging Configuration"
// Structured logging for authentication
func (h *Handler) logAuthEvent(ctx context.Context, event string, userID uuid.UUID, authType string, success bool) {
    logEntry := logrus.WithFields(logrus.Fields{
        "event":       event,
        "user_id":     userID,
        "auth_type":   authType,
        "success":     success,
        "ip":          getClientIP(ctx),
        "user_agent":  getUserAgent(ctx),
        "timestamp":   time.Now().Format(time.RFC3339),
    })
    
    if success {
        logEntry.Info("Authentication event")
    } else {
        logEntry.Warn("Authentication failure")
    }
}
```
